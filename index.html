<script>

    // --- STATE ---

    let trainingPlan = [];

    let chartInstances = {};


    // --- PLAN LOGIC (Same as before) ---

    function generatePlan() {

        const raceDateVal = document.getElementById('raceDateInput').value;

        if (!raceDateVal) return alert("Please select a race date.");

        const raceDate = new Date(raceDateVal);

        trainingPlan = []; 


        for (let w = 19; w >= 0; w--) {

            let weekStart = new Date(raceDate);

            weekStart.setDate(raceDate.getDate() - (w * 7) - (raceDate.getDay() === 0 ? 6 : raceDate.getDay() - 1) - 6);

            

            let phase = w < 2 ? "Taper" : (w < 6 ? "Peak" : (w < 12 ? "Build" : "Base"));

            

            for (let d = 0; d < 7; d++) {

                let currentDay = new Date(weekStart);

                currentDay.setDate(weekStart.getDate() + d);

                const dateStr = currentDay.toISOString().split('T')[0];

                let workouts = [];


                if (d === 1 || d === 4) workouts.push(createWorkout(dateStr, 'swim', phase === 'Base' ? 45 : 60, "Intervals", phase));

                else if (d === 2) workouts.push(createWorkout(dateStr, 'bike', 60, "Zone 3/4", phase));

                else if (d === 3) workouts.push(createWorkout(dateStr, 'run', 45, "Tempo Run", phase));

                else if (d === 5) {

                    let dur = phase === 'Base' ? 90 : (phase === 'Build' ? 120 : 150);

                    if(phase === 'Taper') dur = 60;

                    workouts.push(createWorkout(dateStr, 'bike', dur, "Long Ride", phase));

                } else if (d === 6) {

                    let dur = phase === 'Base' ? 60 : 90;

                    if(phase === 'Taper') dur = 40;

                    workouts.push(createWorkout(dateStr, phase === 'Peak' ? 'brick' : 'run', dur, "Endurance", phase));

                } else if (d === 0) workouts.push(createWorkout(dateStr, 'strength', 30, "Core", phase));


                trainingPlan.push(...workouts);

            }

        }

        renderCalendar();

        updateCharts();

        document.getElementById('exportBtn').disabled = false;

    }


    function createWorkout(date, type, duration, title, phase) {

        return { id: Date.now() + Math.random(), date: date, type: type, duration: duration, title: title, details: `Phase: ${phase}`, metrics: "RPE 5-7" };

    }


    // --- RENDER & CHARTS (Simplified for brevity) ---

    function renderCalendar() {

        const grid = document.getElementById('calendarGrid');

        grid.innerHTML = '';

        if(trainingPlan.length === 0) return;

        trainingPlan.sort((a,b) => new Date(a.date) - new Date(b.date));

        

        const startDate = new Date(trainingPlan[0].date);

        const day = startDate.getDay() || 7; 

        if(day !== 1) startDate.setHours(-24 * (day - 1));


        for(let w=0; w<20; w++) {

            let row = document.createElement('div'); row.className = 'week-row';

            let weekDate = new Date(startDate); weekDate.setDate(startDate.getDate() + (w*7));

            

            let label = document.createElement('div'); label.className = 'week-label';

            label.innerText = `W${w+1}`;

            row.appendChild(label);


            for(let d=0; d<7; d++) {

                let currDate = new Date(weekDate); currDate.setDate(weekDate.getDate() + d);

                let dateStr = currDate.toISOString().split('T')[0];

                let dayCell = document.createElement('div'); dayCell.className = 'day-card';

                dayCell.onclick = () => openAddModal(dateStr);

                

                let dateHeader = document.createElement('div'); dateHeader.className = 'date-header';

                dateHeader.innerText = `${currDate.getDate()}/${currDate.getMonth()+1}`;

                dayCell.appendChild(dateHeader);


                trainingPlan.filter(x => x.date === dateStr).forEach(wo => {

                    let pill = document.createElement('div'); pill.className = `workout-pill type-${wo.type}`;

                    pill.innerHTML = `<b>${wo.type.toUpperCase()}</b> ${wo.duration}m`;

                    pill.onclick = (e) => { e.stopPropagation(); openEditModal(wo.id); };

                    dayCell.appendChild(pill);

                });

                row.appendChild(dayCell);

            }

            grid.appendChild(row);

        }

    }


    function updateCharts() {

        let weeks = Array.from({length: 20}, (_, i) => `W${i+1}`);

        let volumeData = new Array(20).fill(0);

        let typeDist = { swim: 0, bike: 0, run: 0, strength: 0, brick: 0 };

        const startDate = new Date(trainingPlan[0].date);

        

        trainingPlan.forEach(wo => {

            let diff = Math.floor((new Date(wo.date) - startDate) / (1000 * 60 * 60 * 24 * 7));

            if(diff >= 0 && diff < 20) volumeData[diff] += parseInt(wo.duration);

            if(typeDist[wo.type] !== undefined) typeDist[wo.type] += parseInt(wo.duration);

        });


        if(chartInstances.vol) chartInstances.vol.destroy();

        chartInstances.vol = new Chart(document.getElementById('volumeChart'), {

            type: 'bar', data: { labels: weeks, datasets: [{ label: 'Mins', data: volumeData, backgroundColor: '#0f172a' }] }

        });


        if(chartInstances.dist) chartInstances.dist.destroy();

        chartInstances.dist = new Chart(document.getElementById('distChart'), {

            type: 'doughnut', data: { labels: Object.keys(typeDist), datasets: [{ data: Object.values(typeDist), backgroundColor: ['#3b82f6', '#ef4444', '#22c55e', '#a855f7', '#f97316'] }] }

        });

    }


    // --- MODAL FUNCTIONS (Simplified) ---

    function openAddModal(date) { /* setup & show modal */ 

        document.getElementById('modalTitle').innerText = "Add - " + date;

        document.getElementById('editDate').value = date; document.getElementById('editId').value = ''; 

        document.getElementById('editModal').style.display = 'flex';

    }

    function openEditModal(id) { 

        let wo = trainingPlan.find(x => x.id === id);

        document.getElementById('modalTitle').innerText = "Edit";

        document.getElementById('editId').value = id; document.getElementById('editDate').value = wo.date;

        document.getElementById('editType').value = wo.type; document.getElementById('editDuration').value = wo.duration;

        document.getElementById('editMetrics').value = wo.metrics; document.getElementById('editDetails').value = wo.details;

        document.getElementById('editModal').style.display = 'flex';

    }

    function closeModal() { document.getElementById('editModal').style.display = 'none'; }

    

    function saveWorkout() {

        const id = document.getElementById('editId').value;

        const data = {

            date: document.getElementById('editDate').value,

            type: document.getElementById('editType').value,

            duration: parseInt(document.getElementById('editDuration').value),

            metrics: document.getElementById('editMetrics').value,

            details: document.getElementById('editDetails').value,

            title: document.getElementById('editType').value.toUpperCase() + " Session"

        };


        if(id) {

            let idx = trainingPlan.findIndex(x => x.id == id);

            if(idx !== -1) Object.assign(trainingPlan[idx], data);

        } else {

            trainingPlan.push({ ...data, id: Date.now() });

        }

        closeModal(); renderCalendar(); updateCharts();

    }

    function deleteWorkout() {

        let id = document.getElementById('editId').value;

        if(id) trainingPlan = trainingPlan.filter(x => x.id != id);

        closeModal(); renderCalendar(); updateCharts();

    }


    // --- 🚀 NEW: ICS GENERATION LOGIC ---

    function downloadICS() {

        if(trainingPlan.length === 0) return alert("No plan to export!");


        let icsContent = "BEGIN:VCALENDAR\nVERSION:2.0\nPRODID:-//Tri-Dash//Training Plan//EN\nCALSCALE:GREGORIAN\nMETHOD:PUBLISH\n";


        trainingPlan.forEach(wo => {

            // Convert date to YYYYMMDDTHHMMSS format

            // Defaulting all workouts to start at 7:00 AM for consistency

            const year = wo.date.split('-')[0];

            const month = wo.date.split('-')[1];

            const day = wo.date.split('-')[2];

            

            const startStr = `${year}${month}${day}T070000`;

            

            // Calculate end time based on duration

            let startDateObj = new Date(year, month-1, day, 7, 0, 0);

            let endDateObj = new Date(startDateObj.getTime() + wo.duration * 60000);

            

            const endYear = endDateObj.getFullYear();

            const endMonth = String(endDateObj.getMonth() + 1).padStart(2, '0');

            const endDay = String(endDateObj.getDate()).padStart(2, '0');

            const endHour = String(endDateObj.getHours()).padStart(2, '0');

            const endMin = String(endDateObj.getMinutes()).padStart(2, '0');

            

            const endStr = `${endYear}${endMonth}${endDay}T${endHour}${endMin}00`;


            const description = `Details: ${wo.details}\\nMetrics: ${wo.metrics}\\nGenerated by Tri-Dash`;


            icsContent += "BEGIN:VEVENT\n";

            icsContent += `UID:${wo.id}@tridash.local\n`;

            icsContent += `DTSTART:${startStr}\n`;

            icsContent += `DTEND:${endStr}\n`;

            icsContent += `SUMMARY:🏊 ${wo.type.toUpperCase()}: ${wo.duration}m\n`;

            icsContent += `DESCRIPTION:${description}\n`;

            icsContent += "END:VEVENT\n";

        });


        icsContent += "END:VCALENDAR";


        // Trigger Download

        const blob = new window.Blob([icsContent], { type: 'text/calendar' });

        const link = document.createElement('a');

        link.href = window.URL.createObjectURL(blob);

        link.setAttribute('download', 'triathlon-training-plan.ics');

        document.body.appendChild(link);

        link.click();

        document.body.removeChild(link);

    }
